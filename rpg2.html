Как связать с Python‑модулем floor.py

Пример загрузки этажа из JSON и построения объектов Room/Floor в SimplePy: 

import json
from modules.floor import Floor, Room

def load_floor_from_json(path: str) -> Floor:
    with open(path, "r", encoding="utf-8") as f:
        data = json.load(f)

    floor = Floor(data["floor_id"], data["title"])
    room_map = {}

    # Создаём комнаты
    for r in data["rooms"]:
        room = Room(r["id"], r["name"], r["description"])
        room.exits = r.get("exits", {})
        room.loot = r.get("loot", [])
        room.enemy = r.get("enemy")
        floor.add_room(room)
        room_map[r["id"]] = room

    return floor

Для админки (RCB‑Pad++ или админ‑панель на HTML+JS) можно отображать этот JSON в виде сетки карточек (как на странице выше), а при сохранении — отправлять обратно на сервер или в файл.
