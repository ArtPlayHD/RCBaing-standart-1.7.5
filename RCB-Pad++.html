# RCB-Mod: Пример скрипта для Minecraft-RCBaing
# Авто-генерация структуры и спавн сущностей

def generate_structure(x, y, z):
    # Создаем основу
    set_block(x, y, z, "minecraft:stone")
    set_block(x+1, y, z, "minecraft:iron_block")

    # Спавн мобов
    spawn_entity("minecraft:zombie", x, y+1, z)
    print("Структура построена!")

# Основная логика мода
if __name__ == "__main__":
    generate_structure(10, 64, 10)
